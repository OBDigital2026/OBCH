const API_URL = window.location.hostname === 'localhost' ? 'http://localhost:3000/guardar' : 'https://obch.onrender.com/guardar'; form.addEventListener('submit', async (e) => { e.preventDefault(); const numero = document.getElementById('numero').value.trim(); const alfanumerico = document.getElementById('alfanumerico').value.trim(); if (!numero || isNaN(numero) || Number(numero) < 0) { mostrarMensaje('El número debe ser un entero positivo', 'error'); return; } if (!/^[A-Za-z0-9]+$/.test(alfanumerico)) { mostrarMensaje('El campo alfanumérico solo admite letras y números', 'error'); return; } btn.disabled = true; btn.textContent = 'Guardando...'; try { const respuesta = await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ numero: Number(numero), alfanumerico: alfanumerico }) }); const data = await respuesta.json(); if (respuesta.ok) { mostrarMensaje(data.mensaje || 'Hemos recibido su solicitud correctamente', 'exito'); form.reset(); } else { mostrarMensaje(data.error || 'Error al guardar', 'error'); } } catch (err) { mostrarMensaje('No se pudo conectar al servidor. Intenta nuevamente.', 'error'); console.error(err); } finally { btn.disabled = false; btn.textContent = 'Guardar en base de datos'; } });
